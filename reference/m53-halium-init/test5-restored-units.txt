# /usr/lib/systemd/system/no-cpu-hotplug.service
[Unit]
Description=Disable CPU hotplugging during Ubuntu Touch boot
Before=basic.target
After=local-fs.target sysinit.target
DefaultDependencies=no

[Service]
Type=oneshot
ExecStart=:/bin/sh -c 'for cpu in /sys/devices/system/cpu/cpu?/online; do test -e "${cpu}" && chmod 444 "${cpu}"; done'

[Install]
WantedBy=basic.target

# /usr/lib/systemd/system/enable-cpu-hotplug.service
[Unit]
Description=Enable CPU hotplugging after Ubuntu Touch boot

[Service]
Type=oneshot
ExecStart=:/bin/sh -c 'for cpu in /sys/devices/system/cpu/cpu?/online; do test -e "${cpu}" && chmod 644 "${cpu}"; done'
