# /etc/systemd/system/a50-audio-route.service
[Unit]
Description=A50: pin PulseAudio playback to sink.primary-out
Documentation=https://github.com/sadatdaniel/a50-ubuntu-touch/blob/main/docs/experiments/007-abox-firmware-too-early.md
# Must run AFTER the audio HAL has initialised. The HAL applies its own mixer
# settings from /vendor/etc/mixer_paths.xml when PulseAudio loads it, which
# resets ABOX UAIF2 SPK back to RESERVED - so applying the route any earlier
# is silently undone and there is no sound.
After=lxc-android-config.service lightdm.service
Wants=lxc-android-config.service

[Service]
Type=oneshot
RemainAfterExit=yes
# Give PulseAudio time to load the droid card and let the HAL settle before
# overriding the one control it leaves unrouted.
ExecStart=/usr/local/bin/a50-audio-speaker-route.sh
TimeoutStartSec=180

[Install]
WantedBy=multi-user.target
